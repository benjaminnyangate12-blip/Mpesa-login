import os from datetime import datetime from typing import Optional import requests from flask import Flask, render_template, request app = Flask(__name__) MPESA_ENV_KEYS = { 'CONSUMER_KEY': 'MPESA_CONSUMER_KEY', 'CONSUMER_SECRET': 'MPESA_CONSUMER_SECRET', 'SHORTCODE': 'MPESA_SHORTCODE', 'PASSKEY': 'MPESA_PASSKEY', 'CALLBACK_URL': 'MPESA_CALLBACK_URL', 'INITIATOR_NAME': 'MPESA_INITIATOR_NAME', 'SECURITY_CREDENTIAL': 'MPESA_SECURITY_CREDENTIAL', } def get_env(name: str) -> str: value = os.getenv(name, '').strip() if not value: raise RuntimeError(f'Missing environment variable: {name}') return value def get_mpesa_token() -> str: consumer_key = get_env(MPESA_ENV_KEYS['CONSUMER_KEY']) consumer_secret = get_env(MPESA_ENV_KEYS['CONSUMER_SECRET']) url = 'https://sandbox.safaricom.co.ke/oauth/v1/generate?grant_type=client_credentials' response = requests.get(url, auth=(consumer_key, consumer_secret), timeout=30) response.raise_for_status() data = response.json() return data['access_token'] def generate_password(shortcode: str, passkey: str) -> str: timestamp = datetime.now().strftime('%Y%m%d%H%M%S') raw_password = shortcode + passkey + timestamp import base64 return base64.b64encode(raw_password.encode()).decode() def send_stk_push(phone_number: str, amount: str, account_reference: str = 'M-Pesa Login') -> dict: shortcode = get_env(MPESA_ENV_KEYS['SHORTCODE']) passkey = get_env(MPESA_ENV_KEYS['PASSKEY']) callback_url = get_env(MPESA_ENV_KEYS['CALLBACK_URL']) token = get_mpesa_token() timestamp = datetime.now().strftime('%Y%m%d%H%M%S') password = generate_password(shortcode, passkey) payload = { 'BusinessShortCode': shortcode, 'Password': password, 'Timestamp': timestamp, 'TransactionType': 'CustomerPayBillOnline', 'Amount': int(float(amount)), 'PartyA': phone_number, 'PartyB': shortcode, 'PhoneNumber': phone_number, 'CallBackURL': callback_url, 'AccountReference': account_reference, 'TransactionDesc': 'M-Pesa Login Verification', } headers = {'Authorization': f'Bearer {token}', 'Content-Type': 'application/json'} url = 'https://sandbox.safaricom.co.ke/mpesa/stkpush/v1/processrequest' response = requests.post(url, json=payload, headers=headers, timeout=30) response.raise_for_status() return response.json() id_number or not password: return render_template('login.html', error='Please fill in all required fields.') if not phone.isdigit() or len(phone) < 10 or len(phone) > 13: return render_template('login.html', error='Please enter a valid phone number with digits only.') if len(password) < 6: return render_template('login.html', error='Password must be at least 6 characters long.') try: response = send_stk_push(phone, amount='1', account_reference=full_name) except RuntimeError as exc: return render_template('login.html', error=f'M-Pesa is not configured yet: {exc}') except Exception as exc: return render_template('login.html', error=f'Unable to connect to M-Pesa: {exc}') return render_template( 'login.html', success='A payment prompt has been sent to your phone.
